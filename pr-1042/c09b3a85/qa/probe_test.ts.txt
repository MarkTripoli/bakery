async function openImport() {
  const root = await Deno.makeTempDir({ prefix: "bakery-import-execution-" });
  const store = new SkillFileStore(root);
  await store.initialize();
  const fixture = await openAuthFixture({ skillFileStore: store });
  const suffix = crypto.randomUUID();
  const email = `skill-execution-${suffix}@example.test`;
  const password = `SkillExecution-${suffix}`;
  const etag = '"fixture-revision-1"';
  const locator = "https://example.com/skills.tar.gz";
  try {
    await createSuperuser({
      pgDatabase: fixture.pgDatabase,
      config: fixture.config.auth!,
    }, { email, displayName: "Skill Execution Admin", password });
    const scope = await fixture.pgDatabase.query<{ organizationId: string }>(
      `select organization_id as "organizationId"
         from bakery.organization_memberships as membership
         join bakery.principals as principal
           on principal.id = membership.principal_id
         join public."user" as provider_user
           on provider_user.id = principal.provider_user_id
        where provider_user.email = $1`,
      [email],
    );
    const organizationId = scope.rows[0]?.organizationId;
    if (!organizationId) throw new Error("TEST_ORGANIZATION_MISSING");
    const signIn = await fixture.auth.handler(authRequest(
      "/api/auth/sign-in/email",
      { email, password },
    ));
    assertEquals(signIn.status, 200);
    const cookie = sessionCookie(signIn);
    const headers = {
      cookie,
      "Bakery-Organization-Id": organizationId,
    };
    const sourceResponse = await fixture.app.fetch(authRequest(
      "/v1/skill-sources",
      { sourceType: "url", url: locator, name: "Fixture", license: "MIT" },
      {
        cookie,
        headers: { ...headers, "Idempotency-Key": `source-${suffix}` },
      },
    ));
    assertEquals(
      sourceResponse.status,
      201,
      await sourceResponse.clone().text(),
    );
    const source = await sourceResponse.json() as { id: string };
    const importResponse = await fixture.app.fetch(authRequest(
      "/v1/skill-imports",
      { sourceId: source.id, revision: etag },
      {
        cookie,
        headers: { ...headers, "Idempotency-Key": `import-${suffix}` },
      },
    ));
    assertEquals(
      importResponse.status,
      201,
      await importResponse.clone().text(),
    );
    const run = await importResponse.json() as { id: string };
    return {
      root,
      store,
      fixture,
      organizationId,
      headers,
      run,
      etag,
      locator,
      close: async () => {
        await cleanupAuthFixture(fixture.pgDatabase, [email]);
        await fixture.close();
        await Deno.remove(root, { recursive: true });
      },
    };
  } catch (error) {
    await cleanupAuthFixture(fixture.pgDatabase, [email]);
    await fixture.close();
    await Deno.remove(root, { recursive: true });
    throw error;
  }
}

function adapterFor(proof: SkillSourceProof) {
  const requests: SkillSourceFetchRequest[] = [];
  const sourceAdapter: SkillSourceAdapter = {
    fetchPinnedSource(request) {
      requests.push(request);
      return Promise.resolve(proof);
    },
  };
  return { requests, sourceAdapter };
}


const enc = new TextEncoder();
const good = "---\nname: probe\ndescription: A probe skill.\n---\n\n# Probe\n";
const md = (path: string, text: string) => ({ path, content: enc.encode(text), isExecutable: false });
const cases: Record<string, () => Promise<{ manifest: Record<string, unknown>; manifestBytes: Uint8Array; files: ReturnType<typeof md>[] }>> = {
  "manifest.json licence differs from the source": async () => {
    const f = await createRemoteSkillFixture([{ path: "SKILL.md", content: enc.encode(good), mediaType: "text/markdown" }], { license: "Apache-2.0" });
    return { manifest: { ...f.manifest }, manifestBytes: f.manifestBytes, files: f.files.map((x) => ({ path: x.path, content: x.content, isExecutable: false })) };
  },
  "manifest.json does not match a changed file": async () => {
    const f = await createRemoteSkillFixture([{ path: "SKILL.md", content: enc.encode(good), mediaType: "text/markdown" }]);
    return { manifest: { ...f.manifest }, manifestBytes: f.manifestBytes, files: [md("SKILL.md", good + "edited after the manifest\n")] };
  },
  "SKILL.md folder with two names differing only in capitals": async () => ({ manifest: {}, manifestBytes: new Uint8Array(), files: [md("SKILL.md", good), md("Notes.md", "a\n"), md("notes.md", "b\n")] }),
  "SKILL.md folder, valid (control)": async () => ({ manifest: {}, manifestBytes: new Uint8Array(), files: [md("SKILL.md", good)] }),
};
for (const [label, build] of Object.entries(cases)) {
  Deno.test("qa probe: " + label, async () => {
    const { store, fixture, organizationId, run, etag, locator, close } = await openImport();
    try {
      const { sourceAdapter } = adapterFor(Object.freeze({ sourceType: "url", normalizedLocator: locator, requestedRevision: etag, sourceRevision: etag, sourceProofDigest: "d".repeat(64), normalizedPath: "", inventory: Object.freeze(await build()) }));
      const outcome = await executeSkillImport(run.id, { database: fixture.runtimePgDatabase, sourceAdapter, skillFileStore: store }).then((r) => r.state, (e) => "refused " + ((e as { code?: string }).code ?? (e as Error).message));
      const rows = await fixture.pgDatabase.query<{ versions: number; active: boolean }>("select (select count(*)::int from bakery.skill_versions v where v.skill_id = s.id) as versions, s.active_version_id is not null as active from bakery.skills s where s.organization_id = $1 and s.source_id is not null", [organizationId]);
      console.log("PROBE", label, "->", outcome, "| skills left:", rows.rows.length, JSON.stringify(rows.rows));
    } finally { await close(); }
  });
}
